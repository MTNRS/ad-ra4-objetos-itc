```markdown
# Acceso a datos · RA4 · Objetos ITC

Unidad 4: persistencia de objetos. Continúa las entidades clientes y proyectos
de RA2 y RA3 mediante un almacén YAML didáctico. Incluye ventana gráfica.

Instalar: `python -m pip install -r requirements.txt`.
Desde `02-proyecto-unidad`: `python app.py`.
Pruebas: `python -m unittest -v`.

La ventana importa la SQLite creada por C++ en RA2 o utilizada por el ORM de
RA3. Cada cliente se reconstruye como objeto con sus proyectos anidados.
La importación sustituye el almacén YAML tras confirmación y no modifica SQLite.
También permite buscar clientes y cambiar su estado a activo.

YAML permite inspeccionar objetos estructurados con facilidad, pero requiere
cargar el fichero completo y no proporciona índices ni concurrencia. Este
almacén está diseñado para un único proceso. Su transacción guarda una copia
en memoria y sustituye atómicamente el fichero al confirmar. No es una
transacción distribuida entre YAML y SQLite ni un gestor OODB completo.

Es un prototipo local de portabilidad para Integra Tech Consulting con datos
ficticios, sin conexión empresarial. Integrarlo requeriría permisos,
validación del modelo y pruebas en un entorno aislado.
```

```python
import tkinter as tk

ventana=tk.Tk()

almacen.__enter__()
ttk.Label(ventana,text='Clientes y proyectos · Almacén YAML',font=('Segoe UI',18)).pack(pady=15)
busqueda=ttk.Entry(ventana)
busqueda.pack(fill='x',padx=20)
tabla=ttk.Treeview(ventana,columns=('id','nombre','estado','proyectos'),show='headings')
for campo in ('id','nombre','estado','proyectos'): tabla.heading(campo,text=campo.capitalize())
tabla.pack(fill='both',expand=True,padx=20,pady=10)
def refrescar(*args):
    tabla.delete(*tabla.get_children())
    for c in almacen.consultar(busqueda.get()): tabla.insert('', 'end',iid=str(c.id),values=(c.id,c.nombre,c.estado,', '.join(p.nombre for p in c.proyectos)))

    ruta=filedialog.askopenfilename(filetypes=[('SQLite','*.sqlite'),('Todos','*.*')])
    if not ruta:return

        with almacen.transaccion(): almacen.importar_sqlite(ruta)
        refrescar()
    except Exception as e:messagebox.showerror('Importación revertida',str(e))

    seleccion=tabla.selection()
    if not seleccion:return
    with almacen.transaccion():
        for c in almacen.clientes:

botones=ttk.Frame(ventana);botones.pack(pady=15)
for titulo,accion in [('Importar SQLite RA2/RA3',importar),('Activar seleccionado',activar),('Cargar ejemplo ficticio',demo)]:ttk.Button(botones,text=titulo,command=accion).pack(side='left',padx=5)
busqueda.bind('<KeyRelease>',refrescar)
```

**objetos.py**
```python
"""Almacén didáctico de objetos en YAML, para uso local de un proceso."""
from contextlib import contextmanager
from copy import deepcopy
from dataclasses import dataclass, field, asdict
from pathlib import Path
import os
import sqlite3
import tempfile
import yaml

@dataclass
class Proyecto:
    id: int
    nombre: str
    estado: str

@dataclass
class Cliente:
    id: int
    nombre: str
    email: str
    estado: str
    proyectos: list[Proyecto] = field(default_factory=list)

class Almacen:
    def __init__(self, ruta):
        self.ruta = Path(ruta)
        self.clientes = []
        self.abierto = False
        self.en_transaccion = False

    def __enter__(self):
        if self.ruta.exists():
            datos = yaml.safe_load(self.ruta.read_text(encoding='utf-8')) or []
            self.clientes = [Cliente(**{k:v for k,v in x.items() if k!='proyectos'}, proyectos=[Proyecto(**p) for p in x.get('proyectos',[])]) for x in datos]
        self.abierto = True
        return self

    def __exit__(self, *args):
        self.clientes = []
        self.abierto = False

    def consultar(self, texto=''):
        if not self.abierto: raise RuntimeError('Almacén cerrado')
        return deepcopy([c for c in self.clientes if texto.lower() in c.nombre.lower()])

    @contextmanager
    def transaccion(self):
        if not self.abierto or self.en_transaccion: raise RuntimeError('Estado de transacción incorrecto')
        anterior = deepcopy(self.clientes)
        self.en_transaccion = True
        try:
            yield self
            ids = [c.id for c in self.clientes]
            if len(ids)!=len(set(ids)): raise ValueError('Identificadores duplicados')
            self.ruta.parent.mkdir(parents=True, exist_ok=True)
            # Sustitución atómica: un fallo conserva el fichero anterior.
            temporal = None
            try:
                with tempfile.NamedTemporaryFile(mode='w', encoding='utf-8', dir=self.ruta.parent, delete=False) as f:
                    temporal = f.name
                    yaml.safe_dump([asdict(c) for c in self.clientes],f,allow_unicode=True,sort_keys=False)
                    f.flush()
                    os.fsync(f.fileno())
                os.replace(temporal,self.ruta)
            finally:
                if temporal and Path(temporal).exists(): Path(temporal).unlink()
        except Exception:
            self.clientes = anterior
            raise
        finally:
            self.en_transaccion = False

    def importar_sqlite(self, ruta):
        if not self.en_transaccion: raise RuntimeError('Se necesita transacción')
        # Solo lectura sobre la base relacional: no crea ni modifica el origen.
        conexion = sqlite3.connect(Path(ruta).resolve().as_uri()+'?mode=ro',uri=True)
        try:
            objetos = []
            for id,nombre,email,estado in conexion.execute('SELECT id,nombre,email,estado FROM clientes'):
                proyectos = [Proyecto(*p) for p in conexion.execute('SELECT id,nombre,estado FROM proyectos WHERE cliente_id=?',(id,))]
                objetos.append(Cliente(id,nombre,email,estado,proyectos))
            self.clientes = objetos
        finally:
            conexion.close()
```
**test_objetos.py**
```python
import sqlite3
import tempfile
import unittest
from pathlib import Path
from objetos import Almacen, Cliente

class Pruebas(unittest.TestCase):
    def test_persistencia_modificacion_rollback_y_cierre(self):
        with tempfile.TemporaryDirectory() as d:
            ruta=Path(d)/'objetos.yaml'
            with Almacen(ruta) as a:
                with a.transaccion():a.clientes.append(Cliente(1,'Demo','demo@example.test','prospecto'))
                original=ruta.read_bytes()
                with self.assertRaises(ValueError):
                    with a.transaccion():a.clientes.append(Cliente(1,'Duplicado','x@example.test','activo'))
                self.assertEqual(ruta.read_bytes(),original)
                self.assertEqual(len(a.consultar()),1)
                with a.transaccion():a.clientes[0].estado='activo'
            with self.assertRaises(RuntimeError):a.consultar()
            with Almacen(ruta) as b:self.assertEqual(b.consultar('dem')[0].estado,'activo')

    def test_integracion_relacional(self):
        with tempfile.TemporaryDirectory() as d:
            bd=Path(d)/'origen.sqlite'
            c=sqlite3.connect(bd)
            c.executescript("CREATE TABLE clientes(id,nombre,email,estado);CREATE TABLE proyectos(id,cliente_id,nombre,estado);INSERT INTO clientes VALUES(1,'ITC demo','demo@example.test','activo');INSERT INTO proyectos VALUES(2,1,'Portal','pendiente');")
            c.commit();c.close()
            with Almacen(Path(d)/'objetos.yaml') as a:
                with a.transaccion():a.importar_sqlite(bd)
            with Almacen(Path(d)/'objetos.yaml') as a:
                self.assertEqual(a.consultar()[0].proyectos[0].nombre,'Portal')
if __name__=='__main__':unittest.main()
```
